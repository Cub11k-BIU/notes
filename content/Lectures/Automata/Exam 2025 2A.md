# 1
$$
\displaylines{
\text{Given a finite automaton } A: \\
A = \lrp{Q, \Sigma, \delta, q_{0}, F} \\
Q = \Set{ q_{0}, q_{1}, q_{2} } \\
\Sigma = \Set{ a, b } \\
\delta(q_{0}, a) = q_{1} \\
\delta(q_{0}, b) = q_{1} \\
\delta(q_{1}, a) = q_{1} \\
\delta(q_{1}, b) = q_{2} \\
\delta(q_{2}, a) = q_{0} \\
\delta(q_{2}, b) = q_{1} \\
F = \Set{ q_{0}, q_{1} } \\
\text{Build an equivalent right regular CFG as show nin the lectures} \\
\\
\text{Solution:} \\
G = (V, \Sigma, R, S) \\
V = \Set{ S, Q_{0}, Q_{1}, Q_{2} } \\
\Sigma = \Set{ a, b } \\
\begin{align}
S & \to Q_{0} \\
Q_{0} & \to aQ_{1} \mid bQ_{1} \mid \varepsilon \\
Q_{1} & \to aQ_{1} \mid bQ_{2} \mid \varepsilon \\
Q_{2} & \to aQ_{0} \mid bQ_{1} \\
\end{align} \\
}

# 2
$$
\displaylines{
\text{Let } w = \sigma_{1}\dots\sigma_{n} \\
\text{Let } w \mid a^{*} = w_{1}\dots w_{n} \text{ where } \forall i \in [1, n]: w_{i} = \begin{cases}
a^{n_{i}} & \sigma_{i} = a \\
\sigma_{i} & \text{otherwise} \\
\end{cases} \\
\text{In other words, replacement of all } a\text{'s by } a^{*} \\
\text{Let } RR(A) = \bigcup_{w \in A} L(w\mid a^{*}) \\
\text{Prove: } L \text{ is regular} \implies RR(L) \text{ is also regular} \\
\\
\text{Proof:} \\
\text{Let } L \text{ be regular} \\
\text{Let } A = \lrp{Q, \Sigma, \delta, q_{0}, F} \text{ be a DFA such that } L(A) = L \\
\text{We will build } N = \lrp{Q', \Sigma', \Delta', q_{0}', F'} \text{ NFA such that } L(N) = RR(L) \\
\\
\text{To do this, we will have to expand each } a\text{-transition into a separate state (for each)} \\
\text{that will run a loop and then return to the state we initially would've went to} \\
\text{Formally:} \\
Q' = Q \cup \Set{ q_{A} } \times Q \\
\Sigma' = \Sigma \\
q_{0}' = q_{0} \\
F' = F \\
\text{First, copy all transitions except for } a\text{-transitions} \\
\forall q \in Q, \sigma \in \Sigma \setminus \Set{ a }: \Delta'(q, \sigma) = \Set{ \delta(q, \sigma) } \\
\text{Then, replace } a\text{-transitions with a "loop state"} \\
\forall q \in Q: \Delta'(q, \varepsilon) = \Set{ (q_{A}, \delta(q, a)) } \\
\text{In the "loop states" we will simply run a loop of } a\text{'s} \\
\forall p \in Q: \Delta'((q_{A}, p), a) = \Set{ (q_{A}, p) } \\
\text{And return to the state we initially would've went to} \\
\forall p \in Q: \Delta'((q_{A}, p), \varepsilon) = \Set{ p } \\
L(N) = RR(L) \\
\\
\text{Let } r \text{ be a regular expresison such that } L(r) = L \\
\text{Let } R = \Set{ r \mid r \text{ is a regular expression over } \Sigma } \\
\text{Let } f: R \to R, f(r) = r' \text{ where all occurences of } a \text{ are replaced by } a^{*} \\
\text{i.e. } r = (a|b)aba(c)^{*} \implies f(r) = (a^{*}|b)a^{*}ba^{*}(c)^{*} \\
\\
f(r) \in R \text{ and } L(f(r)) = RR(L(r)) = RR(L) \\
\implies RR(L) \text{ is regular} \\
}
$$
# 3
$$
\displaylines{
WA = \Set{ (P_{1}, P_{2}) \mid \exists w \in L(P_{1}): \forall u \in L(P_{2}): \abs{wu} > 3 } \\
\text{Is WA decidable? Is it acceptable?} \\
\\
\text{Solution:} \\
\text{Let us build a reduction from } \overline{ATM} \text{ to } WA \\
\text{We want:} \\
(P, w) \in \overline{ATM} \iff R(P, w) = (P_{1}, P_{2}) \in WA \\
\begin{align}
 & R(P, w): \\
1. & \quad P_{1} = \text{"Q(x): return x == $\varepsilon$"} \\
2. & \quad P_{2} = \text{"Q(x): return P(w) == 1 and |x| <= 3"} \\
3. & \quad \text{return } (P_{1}, P_{2}) \\
\end{align} \\
L(P_{1}) = \Set{ \varepsilon } \\
L(P_{2}) = \begin{cases}
\Set{ x \mid \abs{x} \leq 3 } & P(w) = 1 \\
\emptyset & P(w) \neq 1 \\
\end{cases} \\
(P, w) \in \overline{ATM} \iff P(w) \neq 1 \iff L(P_{2}) = \emptyset \iff (P_{1}, P_{2}) \in WA \\
\implies \boxed{ WA \text{ is not acceptable and not decidable} } \\
}
$$

# 5
$$
\displaylines{
\text{Let } L = \Set{ w \mid w \in \Set{ a, b, c }^{*}, \#a_{w} \geq \#b_{w} } \\
\text{For each following } w \text{ determine whether it proves that } L \text{ is not regular or not} \\
}
$$
### 5a
$$
\displaylines{
w = c^{n-2}a^{2}b^{2} \\
\\
\text{Solution:} \\
\text{Let } x = \varepsilon, y = c^{n-2}, z = a^{2}b^{2} \\
\text{No contradiction} \\
}
$$
### 5b
$$
\displaylines{
w = c^{n-2}a^{2n}b^{n} \\
\\
\text{Solution:} \\
\text{Let } x = \varepsilon, y = c^{n-2}, z = a^{2n}b^{n} \\
\text{No contradiction} \\
}
$$
### 5c
$$
\displaylines{
w = a^{n}c^{n}b^{n} \\
\\
\text{Solution:} \\
\abs{xy} \leq n \implies y = a^{i} \implies xy^{0}z = a^{n-i}c^{n}b^{n} \not\in L \\
\text{Contradiction} \\
}
$$
### 5d
$$
\displaylines{
w = (a^{3}c^{2}b^{1})^{n} \\
\\
\text{Solution:} \\
\text{Let } x = a^{3}, y = c^{2}, z = b(a^{3}c^{2}b^{1})^{n-1} \\
\text{No contradiction} \\
}
$$
### 5e
$$
\displaylines{
w = a^{2n}b^{n}c^{2n} \\
\\
\text{Solution:} \\
\abs{xy} \leq n \implies xy = a^{i}, i \leq n \\
xy^{0}z = a^{2n-i}b^{n}c^{2n} \in L \\
xy^{1+i}z = a^{2n+i}b^{n}c^{2n} \in L \\
\text{No contradiction} \\
}
$$

\text{Let } \Delta(q_{1}, a) = \Set{ q_{1} } \\
\text{Let } \Delta(q_{0}, \varepsilon) = \Set{ q_{1} } \\
\text{Let } \Delta(q_{1}, \varepsilon) = \Set{ q_{0} } \\
\\

$$
\displaylines{
\text{Let } L \\
\text{Let } P \text{ a PDA such that } L(P) = L \\
\text{Then if } P \text{ has at most two states then } L \text{ is regular} \\
\\
\text{Disproof:} \\
\text{Let } L = \Set{ a^{n}b^{k} \mid 0 \leq k \leq n } \\
\text{Let } P = \lrp{Q, \Sigma, \Gamma, \Delta, q_{0}, F} \\
Q = \Set{ q_{0}, q_{1} } \\
\Sigma = \Set{ a, b } \\
\Gamma = \Set{ A } \\
q_{0} = q_{0} \\
F = \Set{ q_{1} } \\
\Delta: \begin{array}{|c|c|}
Q \times \Sigma_{\varepsilon} \times \Gamma_{\varepsilon} & P(Q \times \Gamma^{*}) \\
(q_{0}, a, \varepsilon) & \Set{ (q_{0}, A) } \\
(q_{0}, \varepsilon, \varepsilon) & \Set{ (q_{1}, \varepsilon) } \\
(q_{1}, b, A) & \Set{ (q_{1}, \varepsilon) } \\
\end{array} \\
L(P) = L \\
L \text{ is not regular} \\
}
