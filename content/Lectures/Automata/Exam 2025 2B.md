# 1
$$
\displaylines{
\text{Let there be the following GNFA:} \\
\begin{align}
s & \overset{ a }{ \to } q_{0} \\
s & \overset{ a }{ \to } q_{1} \\
q_{0} & \overset{ b }{ \to } q_{0} \\
q_{0} & \overset{ c }{ \to } q_{1} \\
q_{0} & \overset{ d }{ \to } f \\
q_{1} & \overset{ c }{ \to } q_{0} \\
\end{align} \\
\text{Show an equivalent GNFA without state } q_{0} \text{ using an algorithm learned in lectures} \\
\\
\text{Solution:} \\
\text{Let us build all paths that go through } q_{0} \\
s \overset{ a }{ \to } q_{0} \overset{ b }{ \to } q_{0} \overset{ d }{ \to } f \\
\implies s \overset{ ab^{*}d }{ \to } f \\
s \overset{ a }{ \to } q_{0} \overset{ b }{ \to } q_{0} \overset{ c }{ \to } q_{1} \\
\implies s \overset{ ab^{*}c }{ \to } q_{1} \\
q_{1} \overset{ c }{ \to } q_{0} \overset{ b }{ \to } q_{0} \overset{ d }{ \to } f \\
\implies q_{1} \overset{ cb^{*}d }{ \to } f \\
q_{1} \overset{ c }{ \to } q_{0} \overset{ b }{ \to } q_{0} \overset{ c }{ \to } q_{1} \\
\implies q_{1} \overset{ cb^{*}c }{ \to } q_{1} \\
\\
\text{The resulting GNFA is:} \\
\begin{eqsys}
s \overset{ a \mid ab^{*}c }{ \to } q_{1} \\
s \overset{ ab^{*}d }{ \to } f \\
q_{1} \overset{ cb^{*}c }{ \to } q_{1} \\
q_{1} \overset{ cb^{*}d }{ \to } f \\
\end{eqsys} \\
}
$$
# 2
$$
\displaylines{
\text{Let } w_{-a_{2}} \text{ denote the deletion of every second $a$ from every sequence of $a$'s} \\
\text{i.e. } aaa \to a \cdot a, aaaab \to a \cdot a \cdot b \\
\text{Let } A_{2}(L) = \Set{ w_{-a_{2}} \mid w \in L } \\
\text{Prove: } L \text{ is regular} \implies A_{2}(L) \text{ is regular} \\
\\
\text{Proof:} \\
\text{Let } A = \lrp{Q, \Sigma, \delta, q_{0}, F} \text{ be a DFA such that } L(A) = L \\
\text{We will build } N = \lrp{Q', \Sigma', \Delta', q_{0}', F'} \text{ NFA such that } L(N) = A_{2}(L) \\
\\
\text{To do this we will create a mechanism that will skip every second sequential a} \\
\text{Essentially, there will be two copies of } A, \text{ one that has original (almost) transitions} \\
\text{And another that has only } a\text{-transitions, replaced by } \varepsilon\text{-transitions} \\
\text{Formally:} \\
Q' = Q \times \Set{ 1, 2 } \\
\Sigma' = \Sigma \\
q_{0}' = (q_{0}, 1) \\
F' = F \times \Set{ 1, 2 } \\
\text{First ,copy all non } a\text{-transitions} \\
\forall q \in Q, \sigma \in \Sigma \setminus \Set{ a }: \Delta'((q, 1), \sigma) = \Set{ (\delta(q, \sigma), 1) } \\
\text{And add the same from the second copy back to the original} \\
\forall q \in Q, \sigma \in \Sigma \setminus \Set{ a }: \Delta'((q, 2), \sigma) = \Set{ (\delta(q, \sigma), 1) } \\
\text{Second, redirect } a\text{-transitions to the second copy} \\
\forall q \in Q: \Delta'((q, 1), a) = \Set{ (\delta(q, a), 2) } \\
\text{Finally, add } \varepsilon\text{-transitions that replace second } a\text{'s} \\
\forall p \in Q: \Delta'((p, 2), \varepsilon) = \Set{ (\delta(p, a), 1) } \\
\\
L(N) = A_{2}(L) \\
}
$$
# 3
$$
\displaylines{
\text{Let } WA = \Set{ (P_{1}, P_{2}, w) \mid w \in L(P_{1}) \iff w \in L(P_{2}) } \\
\text{Is } WA \text{ decidable? Is it acceptable?} \\
\\
\text{Solution:} \\
\text{Let us build a computable reduction from } \overline{ATM} \text{ to } WA \\
\text{We want:} \\
(P, w) \in \overline{ATM} \iff R(P, w) = (P_{1}, P_{2}, w) \in WA \\
\begin{align}
 & R(P, w): \\
1. & \quad P_{1} = \text{"Q(x): return P(w) == 1"} \\
2. & \quad P_{2} = \text{"Q(x): return 0"} \\
3. & \quad \text{return } (P_{1}, P_{2}, w) \\
\end{align} \\
L(P_{1}) = \begin{cases}
\Sigma^{*} & P(w) = 1 \\
\emptyset & P(w) \neq 1 \\
\end{cases} \\
L(P_{2}) = \emptyset \\
(P, w) \in \overline{ATM} \iff P(w) \neq 1 \iff L(P_{1}) = \emptyset \iff (P_{1}, P_{2}) \in WA \\
\implies \overline{ATM} \preccurlyeq_{m} WA \implies \boxed{ WA \text{ is not acceptable and not decidable} } \\
}
$$
# 4
$$
\displaylines{
\text{Let } L \subseteq \Set{ a, b, c }^{*} \\
\text{Let for each sequence of } a\text{'s:} \\
\begin{cases}
\text{If there is a b right before the sequence, there is a b right after it (or a)} \\
\text{If there is a c right before the sequence, there is a c right after it (or a)} \\
\end{cases} \\
\text{Build a CFG for } L \\
\\
\text{Solution:} \\
\begin{align}
S & \to aS \mid bT_{b} \mid cT_{c} \mid \varepsilon \\
T_{b} & \to bT_{b} \mid cT_{c} \mid aR_{b} \mid \varepsilon \\
T_{c} & \to cT_{c} \mid bT_{b} \mid aR_{c} \mid \varepsilon \\
R_{b} & \to aR_{b} \mid bT_{b} \mid \varepsilon \\
R_{c} & \to aR_{c} \mid cT_{c} \mid \varepsilon \\
\end{align} \\
}
$$
# Bonus
$$
\displaylines{
\text{Show a non-computable function } f: \Sigma_{1}^{*} \to \Sigma_{2}^{*} \\
\text{Such that } \mathrm{Im}f \text{ is infinite} \\
\\
\text{Solution:} \\
\text{Let } \Sigma_{1}^{*} = \Sigma_{2}^{*} \text{ be a set of all programs, or an empty string} \\
\text{Let } f(P) = \begin{cases}
P & (P, \varepsilon) \in HALT \\
\varepsilon & \text{otherwise} \\
\end{cases} \\
f \text{ is clearly not computable, otherwise HALT is decidable} \\
\\
\mathrm{Im}f = \Set{ P \mid (P, \varepsilon) \in HALT } \cup \Set{ \varepsilon } \text{ which is infinite} \\
}
$$
