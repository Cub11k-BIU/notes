## Variable substitution in integrals #lemma
$$
\displaylines{
\text{When there is a pattern in the integral, being function } f(x) \\
\text{and it's derivative} \\
\text{We will use variable substitution:} \\
\text{Let } t = f(x) \\
t'dt = f'(x)dx \implies dt = f'(x)dx \\
\\
\text{For example:} \\
\int \frac{7\arctan(x)}{x^{2}+1}dx \\
(\arctan(x))' = \frac{1}{x^{2}+1} \\
\implies \int \frac{7\arctan(x)}{x^{2}+1}dx \underbrace{ = }_{ f(x) = \arctan(x) } 7\int f(x)f'(x)dx = \\
\underbrace{ = }_{ t = f(x) } 7\int tdt = \frac{7t^{2}}{2} + C = \frac{7\arctan^{2}(x)}{2} + C \\
\\
\text{Another example:} \\
\int 6\ln(\sin x)\cos^{3}(x)dx \\
\text{Let } t = \sin x \implies dt = \cos(x)dx \\
\implies \int 6\ln(\sin x)\cos^{3}(x)dx = 6 \int \ln(t)(1-t^{2})dt \\
\text{Let } f(t) = \ln(t) \implies f'(t) = \frac{1}{t} \\
\text{Let } g'(t) = 1-t^{2} \implies g(t) = t - \frac{t^{3}}{3} \\
\implies 6 \int \ln(t)(1-t^{2})dt = 6 \left( \ln(t)\left( t-\frac{t^{3}}{3} \right) \right) - 6 \int \left( 1-\frac{t^{2}}{3} \right)dt = \\
= 6t\ln(t) - 2t^{3}\ln(t) - 6t + \frac{2t^{3}}{3} + C = \\
= 6\sin(x)\ln(\sin x) - 2\sin^{3}(x)\ln(\sin x)-6\sin(x) + \frac{2\sin^{3}(x)}{3} + C \\
}
$$
$$
\displaylines{
\text{We can also substitute in a more complex way:} \\
g(t) = f(x) \implies g'(t)dt = f'(x)dx \\
\\
\text{For example:} \\
\int \sqrt{ 7-x^{2} }dx \\
\int \sqrt{ 7-x^{2} }dx = \sqrt{ 7 }\int \sqrt{ 1-\left( \frac{x}{\sqrt{ 7 }} \right)^{2} }dx \\
\text{Let } \sin t = \frac{x}{\sqrt{ 7 }} \implies \cos(t)dt = \frac{dx}{\sqrt{ 7 }} \implies dx = \sqrt{ 7 }\cos(t)dt \\
\implies \sqrt{ 7 }\int \sqrt{ 1-\left( \frac{x}{\sqrt{ 7 }} \right)^{2} }dx = \sqrt{ 7 }\int \sqrt{ 1-\sin^{2}(x) }dx = \sqrt{ 7 }\int \sqrt{ \cos^{2}(x) }dx = \\
= 7\int \cos^{2}(t)dt = 7\int \frac{1+\cos(2t)}{2}dt = \frac{7}{2}\left( t+\frac{\sin(2t)}{2} \right) + C = \\
= \frac{7}{2}\left( \arcsin\left( \frac{x}{\sqrt{ 7 }} \right) + \frac{\sin\left( 2\arcsin\left( \frac{x}{\sqrt{ 7 }} \right) \right)}{2} \right) + C \\
}
$$
## Linear function composition integral #lemma 
$$
\displaylines{
\int f(x)dx = F(x) + C \implies \int f(ax+b)dx = \frac{1}{a}F(ax+b) + C \\
\\
\text{Proof:} \\
\text{Let } t = ax + b \implies dt = adx \\
\implies \int f(ax+b)dx = \int \frac{f(t)}{a}dt = \frac{1}{a}F(t) + C = \frac{1}{a}F(ax+b) + C \\
\\
\text{Example:} \\
\int 3^{x}dx = \frac{3^{x}}{\ln(3)} + C \\
\implies \int 3^{18x-5}dx = \frac{\frac{1}{18}3^{18x-5}}{\ln(3)} + C \\
}
$$
## Integral of a rational function (polynomial fractions) #lemma 
$$
\displaylines{
\int \frac{p(x)}{q(x)}dx \\
\text{There are usually three steps:} \\
1. \quad \text{Division of polynomials} \\
2. \quad \text{Partial fraction decomposition} \\
3. \quad \text{Integration of partial fractions} \\
}
$$
$$
\displaylines{
\text{Division of polynomials:} \\
1.1 \quad \text{Divide highest degree of nominator by highest degree of denominator} \\
1.2 \quad \text{Multiply result by denominator} \\
1.3 \quad \text{Subtract result from nminator} \\
1.4 \quad \text{Repear until degree of nominator is greater or equal to degree of denominator} \\
\\
\text{For example:} \\
\frac{x^{9}+6x^{2}+6}{x^{4}-1} \\
\frac{x^{9}}{x^{4}} = \boxed{ x^{5} } \cdot (x^{4}-1) \implies x^{9}-x^{5} \\
x^{9}+6x^{2}+6 - x^{9}-x^{5} = x^{5}+6x^{2}+6 \\
\frac{x^{5}}{x^{4}} = \boxed{ x } \cdot (x^{4}-1) \implies x^{5}-x \\
x^{5}+6x^{2}+6-x^{5}-x = 6x^{2}+x+6 \\
\implies x^{9}+6x^{2}+6 = x^{5} + x + \frac{6x^{2}+x+6}{x^{4}-1} \\
}
$$
$$
\displaylines{
\text{Partial fraction decomposition:} \\
\text{Any polynomial can be decomposed into product of non-decomposable} \\
\text{polynomials of smaller degree} \\
\text{Non-decomposable polynomial is a polynomial that cannot be represented as a product} \\
\text{of polynomials of smaller degrees} \\
\text{A non-trivial example would be:} \\
x^{4}+1 = x^{4} + 2x^{2} + 1 - 2x^{2} = (x^{2}+1)^{2} - 2x^{2} = (x^{2}+1-\sqrt{ 2 }x)(x^{2}+1+\sqrt{ 2 }x) \\
\text{In our case:} \\
x^{4}-1 = (x^{2}-1)(x^{2}+1) = (x-1)(x+1)(x^{2}+1) \\
\implies \frac{6x^{2}+x+6}{x^{4}-1} = \frac{6x^{2}+x+6}{(x-1)(x+1)(x^{2}+1)} = \frac{.}{x-1} + \frac{.}{x+1} + \frac{.}{x^{2}+1} \\
\text{How do we calculate nominators now?} \\
\text{If denominator is linear, nominator will be a constant} \\
\text{If denominator is of second degree, nominator will be linear} \\
\implies \frac{6x^{2}+x+6}{x^{4}-1} = \frac{A}{x-1} + \frac{B}{x+1} + \frac{Cx+D}{x^{2}+1} \\
6x^{2}+x+6 = A(x+1)(x^{2}+1) + B(x-1)(x^{2}+1) + (Cx+D)(x^{2}-1) \\
\text{Let } x = 1 \\
13 = 4A \implies A = \frac{13}{4} \\
\text{Let } x = -1 \\
11 = -4B \implies B = -\frac{11}{4} \\
A + B + C = 0 \implies C = -\frac{2}{4} \\
}
$$
$$
\displaylines{
\frac{p(x)}{(x-1)^{3}(x+1)} = \frac{Ax^{2}+Bx+C}{(x-1)^{3}} + \frac{D}{x+1} \\
\frac{Ax^{2}-2Ax + A}{(x-1)^{3}} = \frac{A(x-1)^{2}}{(x-1)^{3}} = \frac{A}{x-1} \\
\implies \frac{Ax^{2}+Bx+C}{(x-1)^{3}} = \frac{A}{x-1} + \frac{Bx+2Ax+C-A}{(x-1)^{3}} \\
\frac{Bx + 2Ax - B - 2A}{(x-1)^{3}} = \frac{B+2A}{(x-1)^{2}} \\
\implies \frac{Bx+2Ax+C-A}{(x-1)^{3}} = \frac{B+2A}{(x-1)^{2}} + \frac{-B-C-A}{(x-1)^{3}} \\
\implies \frac{Ax^{2}+Bx + C}{(x-1)^{3}} = \frac{A}{x-1} + \frac{B+2A}{(x-1)^{2}} + \frac{-(A+B+C)}{(x-1)^{3}} = \\
= \frac{A_{1}}{x-1} + \frac{A_{2}}{(x-1)^{2}} + \frac{A_{3}}{(x-1)^{3}} \\
\\
}
$$
#### Table of common denominators in partial fractions #lemma 
$$
\displaylines{
\text{This results in a following table of denominators and their decompositions} \\
\begin{array}{c|c}
\frac{p(x)}{ax+b} & \frac{A}{ax + b} \\
\frac{p(x)}{(ax+b)^{k}} & \sum_{i=1}^{k} \frac{A_{i}}{(ax+b)^{i}} \\
\frac{p(x)}{ax^{2}+bx+c} & \frac{Ax+B}{ax^{2}+bx+c} \\
\frac{p(x)}{(ax^{2}+bx+c)^{k}} & \sum_{i=1}^{k} \frac{A_{i}x+B_{i}}{(ax^{2}+bx+c)^{i}} \\
\end{array} \\
}
$$
$$
\displaylines{
\text{Integration of partial fractions:} \\
\int \frac{A}{ax+b}dx = \frac{A}{a}\ln \lvert ax+b \rvert + C \\
\int \frac{A}{(ax+b)^{k}}dx = \frac{A(ax+b)^{1-k}}{1-k} + C \\

\int \frac{p'(x)}{p(x)}dx = \int \frac{1}{p}dp = \ln \lvert p(x) \rvert + C \\
}
