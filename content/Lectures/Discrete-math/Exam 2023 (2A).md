$$
\displaylines{
\text{Let } G = (V, E) \text{ be a simple non-empty finite connected graph} \\
\text{Let } e \in E : G' = (V, E \setminus \Set{ e }) \text{ has two connected components} \\
\text{Prove: } \exists v \in V: deg(v) \text{ is odd} \\
\\
\text{Proof:} \\
\text{Let } \forall v \in V: deg(v) \text{ is even} \\
\text{Let } e = \Set{ v, u } \\
deg(v) \text{ in } G' \text{ is equal to } deg(v) \text{ in } G \implies \text{it is odd} \\
\sum_{u \in [w]_{\sim}} deg(w) = 2 \cdot \lvert E_{v} \rvert = \underbrace{ \sum_{u \in [w]_{\sim} \setminus \Set{ v }} \underbrace{ deg(w) }_{ \text{even} } }_{ \text{even} } + \underbrace{ deg(v) }_{ odd } \\
\implies 2 \cdot \lvert E_{v} \rvert \text{ is odd} - \text{Contradiction!} \\
\implies \boxed{ \exists v \in V: deg(v) \text{ is odd} } \\
}

# 2b
$$
\displaylines{
\text{Let } A, B \text{ be sets} \\
\text{Prove or disprove: } P(A) \cup P(B) = P(A \cup B) \iff (A \subseteq B) \lor (B \subseteq A) \\
\\
\text{Proof:} \\
\text{Let } B \neq A \cup B \neq A \\
\implies A \cup B \in P(A \cup B), A \cup B \not\in P(A), A \cup B \not\in P(B) \\
\implies P(A) \cup P(B) \neq P(A \cup B) \\
\implies P(A) \cup P(B) = P(A \cup B) \implies (\underbrace{ A \cup B = A }_{ B \subseteq A }) \lor (\underbrace{ A \cup B = B }_{ A \subseteq B }) \\
\impliedby \text{ is trivial} \\
}
$$

# 5
$$
\displaylines{
\forall a \in \mathbb{R}: \forall r \geq 0 \in \mathbb{R} \text{ define an open-ball of radius } r \text{ around } a \text{ to be} \\
B(a, r) = \Set{ x \in \mathbb{R} | \lvert x - a \rvert < r } \\
\text{Set } U \subseteq \mathbb{R} \text{ is called open if } \forall a \in U: \exists B(a, r) \subseteq U \\
\text{Set } X \text{ is called closed if } X^{c} \text{ is open} \\
}
$$
## 5a
$$
\displaylines{
\text{Is there a subset of } \mathbb{R} \text{ which is both open and closed?} \\
\\
\text{Solution:} \\
\emptyset \text{ is open (vacuously true)} \\
\emptyset^{c} = \mathbb{R} \text{ is open, } B(x, 1) \subseteq \mathbb{R} \\
\implies \emptyset \text{ is also closed} \\
}
$$
## 5b
$$
\displaylines{
\text{Let } \Set{ U_{i} }_{i \in [n]} \text{ be family of sets} \\
\text{Prove: } \forall i \in [n]: U_{i} \text{ is open} \implies \bigcap_{i=1}^{n} U_{i} \text{ is open} \\
\\
\text{Proof:} \\
\text{Let } x \in \bigcap_{i=1}^{n} U_{i} \\
\implies \forall i \in [n]: x \in U_{i} \implies \forall i \in [n]: \exists B(x, r_{i}) \subseteq U_{i} \\
n \in \mathbb{N} \implies B = \Set{ r_{1}, r_{2}, \dots, r_{n} } \text{ is finite} \\
\implies \exists r \in B: \forall r_{i} \in B: r_{i} \leq r \implies r_{i} = r \\
\forall i \in [n]: r \leq r_{i} \implies B(x, r) \subseteq U_{i} \\
\implies \boxed{ B(x, r) \subseteq \bigcap_{i=1}^{n} U_{i} } \\
}
$$
